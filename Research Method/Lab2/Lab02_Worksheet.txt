## Exercise 3 answer:


ggplot(my_data, aes(x = factor(cyl), y = hp)) + 
  geom_bar(stat = "summary", fun = "mean", fill = "green", color = "black") +
  labs(title = "Average Horsepower by Cylinder Count",
       x = "Number of Cylinders",
       y = "Average Horsepower")

ggplot(my_data, aes(x = cyl, y = hp)) + 
  geom_boxplot(aes(group = cyl), fill = "blue", color = "black") +
  labs(title = "Horsepower Distribution by Cylinder Count",
       x = "Number of Cylinders",
       y = "Horsepower (hp)")
